# Enter your code here. Read input from STDIN. Print output to STDOUT

n = int(input())

words = {}
order = []

for _ in range(n):
    word = input().strip()

    if word in words:
        words[word] += 1
    else:
        words[word] = 1
        order.append(word)

print(len(words))
print(*[words[word] for word in order])
